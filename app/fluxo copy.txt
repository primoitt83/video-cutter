1. Funcionalidades Implementadas
Upload de Vídeo

    O usuário pode enviar um vídeo para o servidor.

    O vídeo é armazenado na pasta uploads no servidor.

    O tamanho máximo permitido é 100 MB.

    Após o upload, o vídeo é exibido em um player para visualização.

Corte de Vídeo

    O usuário pode selecionar os tempos de início e fim para cortar o vídeo.

    Há uma opção de corte rápido (sem re-encode) para maior eficiência.

    O vídeo cortado é salvo na pasta outputs e disponibilizado para download.

Recorte (Crop) de Vídeo

    O usuário pode selecionar uma área do vídeo para recortar.

    A seleção é feita em um canvas sobreposto ao vídeo.

    O vídeo recortado é salvo na pasta outputs e disponibilizado para download.

Ambos (Corte e Recorte)

    O usuário pode aplicar corte e recorte ao mesmo tempo.

    As operações são realizadas em um único comando FFmpeg, sem a necessidade de criar arquivos intermediários.

Remoção de Vídeos Após Processamento

    O usuário pode optar por remover os vídeos (original e processado) após o processamento.

    Se a opção estiver marcada:

        O vídeo original é excluído após o processamento.

        O vídeo processado é excluído após o download.

2. Fluxo de Trabalho
Passo a Passo

    Upload do Vídeo:

        O usuário seleciona um vídeo e faz o upload.

        O vídeo é exibido no player.

    Seleção de Função:

        O usuário escolhe entre Cortar Vídeo, Recortar (Crop) Vídeo ou Ambos.

        Há também a opção de Remover vídeos após processar.

    Corte de Vídeo:

        O usuário seleciona os tempos de início e fim.

        Confirma o corte, e uma mensagem de "Confirmado" é exibida.

    Recorte (Crop) de Vídeo:

        O usuário seleciona a área de recorte no canvas.

        Confirma o recorte, e uma mensagem de "Confirmado" é exibida.

    Processamento:

        O usuário clica em Processar Vídeo.

        Uma mensagem de "Processando... aguarde..." é exibida.

        Após o processamento, o botão Baixar Vídeo Processado é exibido.

    Download:

        O usuário baixa o vídeo processado.

        Se a opção de remoção estiver ativa, o vídeo é excluído após o download.

server.js
const express = require('express');
const multer = require('multer');
const path = require('path');
const fs = require('fs');
const ffmpeg = require('fluent-ffmpeg');

const app = express();
const port = 3000;

// Configurar pastas
const uploadDir = path.join(__dirname, 'uploads');
const outputDir = path.join(__dirname, 'outputs');

if (!fs.existsSync(uploadDir)) fs.mkdirSync(uploadDir);
if (!fs.existsSync(outputDir)) fs.mkdirSync(outputDir);

const storage = multer.diskStorage({
  destination: (req, file, cb) => cb(null, uploadDir),
  filename: (req, file, cb) => cb(null, `${Date.now()}-${Math.random().toString(36).substring(7)}.mp4`),
});
const upload = multer({ storage });

app.use('/uploads', express.static(uploadDir));
app.use('/outputs', express.static(outputDir));
app.use(express.json());

// Rota para upload de vídeo
app.post('/upload', upload.single('video'), (req, res) => {
  res.json({ filename: req.file.filename, path: `/uploads/${req.file.filename}` });
});

// Rota para processar vídeo (corte e/ou recorte)
app.post('/process-video', (req, res) => {
  const { startTime, endTime, filename, fastCut, x, y, width, height, removeAfterProcessing } = req.body;
  const inputPath = path.join(uploadDir, filename);
  const outputFilename = `${Date.now()}_processed.mp4`;
  const outputPath = path.join(outputDir, outputFilename);

  const command = ffmpeg(inputPath);

  // Aplica o corte, se necessário
  if (startTime !== undefined && endTime !== undefined) {
    const duration = endTime - startTime;
    command.setStartTime(startTime).setDuration(duration);
  }

  // Aplica o recorte, se necessário
  if (width && height) {
    command.videoFilter(`crop=${width}:${height}:${x}:${y}`);
  }

  // Aplica o corte rápido, se necessário
  if (fastCut) {
    command.outputOptions('-c', 'copy');
  }

  command
    .output(outputPath)
    .on('end', () => {
      // Remove o vídeo original, se a opção estiver marcada
      if (removeAfterProcessing) {
        fs.unlinkSync(inputPath); // Remove o vídeo original
      }

      // Retorna o URL do vídeo processado
      res.json({ url: `/outputs/${outputFilename}`, removeAfterProcessing });
    })
    .on('error', (err) => {
      console.error('Erro ao processar vídeo:', err.message);
      res.status(500).json({ error: 'Erro ao processar o vídeo', details: err.message });
    })
    .run();
});

// Rota para baixar o vídeo processado
app.get('/download-processed', (req, res) => {
  const { filename, removeAfterDownload } = req.query;
  const filePath = path.join(outputDir, filename);

  // Verifica se o arquivo existe
  if (!fs.existsSync(filePath)) {
    return res.status(404).json({ error: 'Arquivo não encontrado' });
  }

  // Envia o arquivo para o cliente
  res.download(filePath, (err) => {
    if (err) {
      console.error('Erro ao enviar o arquivo:', err.message);
      return res.status(500).json({ error: 'Erro ao enviar o arquivo' });
    }

    // Remove o vídeo processado após o download, se a opção estiver marcada
    if (removeAfterDownload === 'true') {
      fs.unlinkSync(filePath);
    }
  });
});

// Servir o index.html na raiz
app.get('/', (req, res) => {
  res.sendFile(path.join(__dirname, 'index.html'));
});

app.listen(port, () => {
  console.log(`Servidor rodando na porta ${port}`);
});

    <script>
        let uploadedFilename = '';
        let startTime = 0;
        let endTime = 0;
        let isCropping = false;
        let cropCoords = { x: 0, y: 0, width: 0, height: 0 };
        let isDragging = false;
        let startX, startY;
        let videoWidth, videoHeight;
        let canvasWidth, canvasHeight;
        let scale = 1;
        let selectedFunction = '';
        let removeAfterProcessing = false;

        document.getElementById('uploadForm').addEventListener('submit', async function(e) {
            e.preventDefault();
            const fileInput = document.getElementById('videoInput');
            if (fileInput.files[0].size > 100 * 1024 * 1024) { // 100 MB
                showError('O arquivo é muito grande. O tamanho máximo permitido é 100 MB.');
                return;
            }

            const formData = new FormData();
            formData.append('video', fileInput.files[0]);

            const progressBar = document.querySelector('#progressBar div');
            progressBar.style.width = '0%';

            const xhr = new XMLHttpRequest();
            xhr.open('POST', '/upload', true);

            xhr.upload.onprogress = function(event) {
                if (event.lengthComputable) {
                    const percentComplete = (event.loaded / event.total) * 100;
                    progressBar.style.width = percentComplete + '%';
                }
            };

            xhr.onload = function() {
                if (xhr.status === 200) {
                    const data = JSON.parse(xhr.responseText);
                    uploadedFilename = data.filename;

                    const videoPlayer = document.getElementById('videoPlayer');
                    videoPlayer.src = `/uploads/${uploadedFilename}`;
                    videoPlayer.style.display = 'block';

                    videoPlayer.onloadedmetadata = () => {
                        videoWidth = videoPlayer.videoWidth;
                        videoHeight = videoPlayer.videoHeight;
                        setupCanvas();
                    };

                    // Mostra as opções de função após o upload
                    document.getElementById('functionSelector').classList.remove('hidden');
                }
            };

            xhr.onerror = function() {
                showError('Erro ao enviar o vídeo. Tente novamente.');
            };

            xhr.send(formData);
        });

        function setupCanvas() {
            const videoCanvas = document.getElementById('videoCanvas');
            const aspectRatio = videoWidth / videoHeight;

            // Define o tamanho do canvas com base na proporção do vídeo
            canvasWidth = 640; // Largura fixa para o canvas
            canvasHeight = canvasWidth / aspectRatio;

            videoCanvas.width = canvasWidth;
            videoCanvas.height = canvasHeight;
            videoCanvas.style.display = 'block';

            // Calcula a escala para mapear as coordenadas do canvas para o vídeo
            scale = videoWidth / canvasWidth;
        }

        function selectFunction(functionType) {
            selectedFunction = functionType;
            document.getElementById('functionSelector').classList.add('hidden');

            if (functionType === 'cut' || functionType === 'both') {
                document.getElementById('cutControls').classList.remove('hidden');
            }

            if (functionType === 'crop' || functionType === 'both') {
                if (functionType === 'both') {
                    // No modo "Ambos", os controles de recorte só aparecem após confirmar o corte
                    document.getElementById('cropControls').classList.add('hidden');
                } else {
                    document.getElementById('cropControls').classList.remove('hidden');
                }
            }
        }

        function setStartTime() {
            startTime = document.getElementById('videoPlayer').currentTime;
            document.getElementById('startTimeLabel').innerText = `Início: ${startTime.toFixed(2)}s`;
        }

        function setEndTime() {
            endTime = document.getElementById('videoPlayer').currentTime;
            document.getElementById('endTimeLabel').innerText = `Fim: ${endTime.toFixed(2)}s`;
        }

        function confirmCut() {
            if (startTime >= endTime) {
                showError('O tempo de início deve ser menor que o tempo de fim.');
                return;
            }

            // Mostra a mensagem de confirmação
            document.getElementById('cutConfirmMessage').classList.remove('hidden');

            if (selectedFunction === 'cut') {
                // Mostra o botão de processamento
                document.getElementById('processButton').classList.remove('hidden');
            } else if (selectedFunction === 'both') {
                // No modo "Ambos", mostra os controles de recorte após confirmar o corte
                document.getElementById('cropControls').classList.remove('hidden');
            }
        }

        function startCrop() {
            isCropping = true;
            const videoCanvas = document.getElementById('videoCanvas');
            videoCanvas.style.display = 'block';
            document.getElementById('cropBox').classList.remove('hidden');
            drawVideoOnCanvas();
        }

        function drawVideoOnCanvas() {
            const videoCanvas = document.getElementById('videoCanvas');
            const ctx = videoCanvas.getContext('2d');
            const videoPlayer = document.getElementById('videoPlayer');
            ctx.clearRect(0, 0, canvasWidth, canvasHeight);
            ctx.drawImage(videoPlayer, 0, 0, canvasWidth, canvasHeight);
            if (isCropping) {
                requestAnimationFrame(drawVideoOnCanvas);
            }
        }

        function confirmCrop() {
            isCropping = false;
            const cropBox = document.getElementById('cropBox');
            cropCoords = {
                x: parseInt(cropBox.style.left) * scale,
                y: parseInt(cropBox.style.top) * scale,
                width: parseInt(cropBox.style.width) * scale,
                height: parseInt(cropBox.style.height) * scale
            };

            // Mostra a mensagem de confirmação
            document.getElementById('cropConfirmMessage').classList.remove('hidden');

            // Mostra o botão de processamento
            document.getElementById('processButton').classList.remove('hidden');
        }

        async function processVideo() {
            const fastCut = document.getElementById('fastCutCheckbox').checked;
            removeAfterProcessing = document.getElementById('removeAfterProcessing').checked;

            // Mostra a mensagem de processamento
            document.getElementById('processingMessage').classList.remove('hidden');

            const response = await fetch('/process-video', {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify({
                    startTime: selectedFunction === 'crop' ? undefined : startTime,
                    endTime: selectedFunction === 'crop' ? undefined : endTime,
                    filename: uploadedFilename,
                    fastCut,
                    x: cropCoords.x,
                    y: cropCoords.y,
                    width: cropCoords.width,
                    height: cropCoords.height,
                    removeAfterProcessing
                })
            });

            if (!response.ok) {
                const errorData = await response.json();
                showError(errorData.error || 'Erro ao processar o vídeo.');
                return;
            }

            const data = await response.json();

            // Esconde a mensagem de processamento e mostra o botão de download
            document.getElementById('processingMessage').classList.add('hidden');
            const downloadLink = document.getElementById('downloadLink');
            downloadLink.href = `/download-processed?filename=${data.url.split('/').pop()}&removeAfterDownload=${removeAfterProcessing}`;
            downloadLink.classList.remove('hidden');
            downloadLink.innerText = 'Baixar Vídeo Processado';
        }

        function showError(message) {
            const errorMessage = document.getElementById('errorMessage');
            errorMessage.innerText = message;
            setTimeout(() => errorMessage.innerText = '', 5000);
        }

        // Interação com a caixa de recorte
        const videoCanvas = document.getElementById('videoCanvas');
        const cropBox = document.getElementById('cropBox');

        videoCanvas.addEventListener('mousedown', (e) => {
            if (!isCropping) return;
            isDragging = true;
            startX = e.offsetX;
            startY = e.offsetY;
            cropBox.style.left = `${startX}px`;
            cropBox.style.top = `${startY}px`;
            cropBox.style.width = '0px';
            cropBox.style.height = '0px';
        });

        videoCanvas.addEventListener('mousemove', (e) => {
            if (!isDragging) return;
            const width = e.offsetX - startX;
            const height = e.offsetY - startY;
            cropBox.style.width = `${Math.abs(width)}px`;
            cropBox.style.height = `${Math.abs(height)}px`;
            cropBox.style.left = `${width > 0 ? startX : e.offsetX}px`;
            cropBox.style.top = `${height > 0 ? startY : e.offsetY}px`;
        });

        videoCanvas.addEventListener('mouseup', () => {
            isDragging = false;
        });
    </script>
